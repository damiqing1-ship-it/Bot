    function calculateSMA(prices, period) {
        if (prices.length < period) return null;
        return prices.slice(-period).reduce((a, b) => a + b, 0) / period;
    }

    function calculateEMA(prices, period) {
        if (prices.length < period) return null;
        const multiplier = 2 / (period + 1);
        let ema = calculateSMA(prices.slice(0, period), period);
        for (let i = period; i < prices.length; i++) {
            ema = (prices[i] - ema) * multiplier + ema;
        }
        return ema;
    }

    function calculateStdDev(prices, period) {
        if (prices.length < period) return null;
        const slice = prices.slice(-period);
        const mean = slice.reduce((a, b) => a + b, 0) / period;
        const variance = slice.reduce((sum, val) => sum + Math.pow(val - mean, 2), 0) / period;
        return Math.sqrt(variance);
    }

    function mean(arr) {
        return arr.reduce((a, b) => a + b, 0) / arr.length;
    }

    function stdDev(arr) {
        const m = mean(arr);
        const variance = arr.reduce((sum, val) => sum + Math.pow(val - m, 2), 0) / arr.length;
        return Math.sqrt(variance);
    }

    function covariance(arr1, arr2) {
        const m1 = mean(arr1);
        const m2 = mean(arr2);
        let sum = 0;
        for (let i = 0; i < arr1.length; i++) {
            sum += (arr1[i] - m1) * (arr2[i] - m2);
        }
        return sum / arr1.length;
    }

    function calculateRSI(prices, period = 14) {
        if (prices.length < period + 1) return 50;
        let gains = 0, losses = 0;
        for (let i = prices.length - period; i < prices.length; i++) {
            const change = prices[i] - prices[i - 1];
            if (change > 0) gains += change;
            else losses += Math.abs(change);
        }
        const avgGain = gains / period;
        const avgLoss = losses / period;
        if (avgLoss === 0) return 100;
        const rs = avgGain / avgLoss;
        return 100 - (100 / (1 + rs));
    }

    function calculateMACD(prices) {
        if (prices.length < 26) return { macd: 0, signal: 0, histogram: 0 };
        const ema12 = calculateEMA(prices, 12);
        const ema26 = calculateEMA(prices, 26);
        const macd = ema12 - ema26;
        const signal = calculateEMA([macd], 9) || macd;
        return { macd, signal, histogram: macd - signal };
    }

    function calculateCorrelation(prices1, prices2, period = 100) {
        if (prices1.length < period || prices2.length < period) return 0;
        const slice1 = prices1.slice(-period);
        const slice2 = prices2.slice(-period);
        const cov = covariance(slice1, slice2);
        const std1 = stdDev(slice1);
        const std2 = stdDev(slice2);
        if (std1 === 0 || std2 === 0) return 0;
        return cov / (std1 * std2);
    }

    function calculateZScore(prices1, prices2, beta, period = 100) {
        if (prices1.length < period || prices2.length < period) return 0;
        const spreads = [];
        const minLen = Math.min(prices1.length, prices2.length);
        for (let i = 0; i < minLen; i++) {
            spreads.push(prices1[i] - beta * prices2[i]);
        }
        const recentSpreads = spreads.slice(-period);
        const spreadMean = mean(recentSpreads);
        const spreadStd = stdDev(recentSpreads);
        if (spreadStd === 0) return 0;
        return (spreads[spreads.length - 1] - spreadMean) / spreadStd;
    }

    function calculateVolatility(prices, period = 20) {
        if (prices.length < period) return 0;
        const returns = [];
        for (let i = prices.length - period; i < prices.length - 1; i++) {
            returns.push((prices[i + 1] - prices[i]) / prices[i]);
        }
        return calculateStdDev(returns, returns.length) || 0;
    }

    function detectTrend(prices) {
        if (prices.length < 50) return 0;
        const shortSMA = calculateSMA(prices, 20);
        const longSMA = calculateSMA(prices, 50);
        if (!shortSMA || !longSMA) return 0;
        if (shortSMA > longSMA * 1.015) return 1;
        if (shortSMA < longSMA * 0.985) return -1;
        return 0;
    }

    function calculateBollingerBands(prices, period = 20) {
        const sma = calculateSMA(prices, period);
        const stdDev = calculateStdDev(prices, period);
        if (!sma || !stdDev) return { upper: 0, middle: 0, lower: 0 };
        return {
            upper: sma + (2 * stdDev),
            middle: sma,
            lower: sma - (2 * stdDev)
        };
    }

    function calculateCointegration(prices1, prices2) {
        if (prices1.length < 60 || prices2.length < 60) return 1;
        const n = Math.min(prices1.length, prices2.length);
        const slice1 = prices1.slice(-n);
        const slice2 = prices2.slice(-n);
        const meanX = mean(slice2);
        const meanY = mean(slice1);
        let numerator = 0, denominator = 0;
        for (let i = 0; i < n; i++) {
            numerator += (slice2[i] - meanX) * (slice1[i] - meanY);
            denominator += Math.pow(slice2[i] - meanX, 2);
        }
        const beta = denominator !== 0 ? numerator / denominator : 1;
        const residuals = [];
        for (let i = 0; i < n; i++) {
            residuals.push(slice1[i] - beta * slice2[i]);
        }
        const residualStd = stdDev(residuals);
        let adfStat = 0;
        for (let i = 1; i < residuals.length; i++) {
            adfStat += Math.abs(residuals[i] - residuals[i-1]);
        }
        adfStat = adfStat / (residuals.length * residualStd);
        return Math.min(1, adfStat / 10);
    }

    function calculateKalmanBeta(prices1, prices2) {
        if (prices1.length < 20 || prices2.length < 20) return 1;
        let beta = 1, P = 1;
        const Q = 0.0001, R = 0.1;
        const n = Math.min(prices1.length, prices2.length);
        const windowSize = Math.min(50, n);
        for (let i = n - windowSize; i < n; i++) {
            const P_pred = P + Q;
            const K = P_pred / (P_pred + R);
            const measurement = prices2[i] !== 0 ? prices1[i] / prices2[i] : beta;
            beta = beta + K * (measurement - beta);
            P = (1 - K) * P_pred;
        }
        return beta;
    }

    function calculateOUTheta(spreads) {
        if (spreads.length < 50) return 0;
        const recentSpreads = spreads.slice(-100);
        const spreadMean = mean(recentSpreads);
        let sumDiff = 0, sumSquaredDiff = 0;
        for (let i = 1; i < recentSpreads.length; i++) {
            const diff = recentSpreads[i] - recentSpreads[i-1];
            const deviation = recentSpreads[i-1] - spreadMean;
            sumDiff += diff * deviation;
            sumSquaredDiff += Math.pow(deviation, 2);
        }
        return sumSquaredDiff !== 0 ? Math.max(0, -sumDiff / sumSquaredDiff) : 0;
    }

    function calculateHalfLife(theta) {
        return theta <= 0 ? 0 : Math.log(2) / theta;
    }

    function calculateHurst(prices) {
        if (prices.length < 100) return 0.5;
        const recentPrices = prices.slice(-100);
        const n = recentPrices.length;
        const returns = [];
        for (let i = 1; i < n; i++) {
            returns.push(Math.log(recentPrices[i] / recentPrices[i-1]));
        }
        const returnMean = mean(returns);
        const cumDev = [];
        let sum = 0;
        for (let i = 0; i < returns.length; i++) {
            sum += returns[i] - returnMean;
            cumDev.push(sum);
        }
        const range = Math.max(...cumDev) - Math.min(...cumDev);
        const std = stdDev(returns);
        if (std === 0) return 0.5;
        const rs = range / std;
        return Math.max(0, Math.min(1, Math.log(rs) / Math.log(n)));
    }

    function calculateOrderFlow(prices) {
        if (prices.length < 20) return 0;
        const recentPrices = prices.slice(-20);
        let buyVolume = 0, sellVolume = 0;
        for (let i = 1; i < recentPrices.length; i++) {
            const priceChange = recentPrices[i] - recentPrices[i-1];
            const volume = Math.abs(priceChange);
            if (priceChange > 0) buyVolume += volume;
            else if (priceChange < 0) sellVolume += volume;
        }
        const totalVolume = buyVolume + sellVolume;
        return totalVolume === 0 ? 0 : (buyVolume - sellVolume) / totalVolume;
    }

    function calculateVWAP(prices) {

        for (let i = 0; i < recentPrices.length; i++) {
            sumPriceVolume += recentPrices[i];
            sumVolume += 1;
        }
        return sumVolume !== 0 ? sumPriceVolume / sumVolume : recentPrices[recentPrices.length - 1];
    }

    function calculateVWAPDeviation(currentPrice, vwap) {
        return vwap === 0 ? 0 : ((currentPrice - vwap) / vwap) * 100;
    }

(function() {
    'use strict';

    const P1 = window.STARB_PART1;

    function checkMomentum(prices, bars = 3) {
        if (prices.length < bars + 1) return 0;
        let upBars = 0, downBars = 0;
        for (let i = 0; i < bars; i++) {
            const current = prices[prices.length - 1 - i];
            const previous = prices[prices.length - 2 - i];
            if (current > previous) upBars++;
            else if (current < previous) downBars++;
        }
        if (upBars === bars) return 1;
        if (downBars === bars) return -1;
        return 0;
    }

    function generateBinarySignal(pair, data) {
        if (!P1.isPairEnabled(pair.id)) {
            return { signal: 'disabled', reason: 'Pair disabled', confidence: 0, zScore: 0, correlation: 0, rsi1: 50, rsi2: 50, timestamp: new Date() };
        }

        const { price1, price2, prices1, prices2 } = data;
        if (prices1.length < P1.CONFIG.minDataPoints || prices2.length < P1.CONFIG.minDataPoints) {
            return { signal: 'none', reason: 'Insufficient data', confidence: 0, timestamp: new Date() };
        }

        const now = Date.now();
        if (P1.lastSignalTime[pair.id] && (now - P1.lastSignalTime[pair.id]) < P1.CONFIG.signalCooldown) {
            const remainingTime = Math.ceil((P1.CONFIG.signalCooldown - (now - P1.lastSignalTime[pair.id])) / 1000);
            return { signal: 'cooldown', reason: `Next signal in ${remainingTime}s`, confidence: 0, timestamp: new Date() };
        }

        const kalmanBeta = P1.calculateKalmanBeta(prices1, prices2);
        const cointegrationPValue = P1.calculateCointegration(prices1, prices2);
        const zScore = P1.calculateZScore(prices1, prices2, kalmanBeta, 100);
        const correlation = P1.calculateCorrelation(prices1, prices2, 100);
        const hurst = P1.calculateHurst(prices1);

        const spreads = [];
        for (let i = 0; i < Math.min(prices1.length, prices2.length); i++) {
            spreads.push(prices1[i] - kalmanBeta * prices2[i]);
        }

        const ouTheta = P1.calculateOUTheta(spreads);
        const halfLife = P1.calculateHalfLife(ouTheta);
        const orderFlow1 = P1.calculateOrderFlow(prices1);
        const orderFlow2 = P1.calculateOrderFlow(prices2);
        const netOrderFlow = orderFlow1 - orderFlow2;
        const vwap1 = P1.calculateVWAP(prices1);
        const vwapDev = P1.calculateVWAPDeviation(price1, vwap1);

        const rsi1 = P1.calculateRSI(prices1, 14);
        const rsi2 = P1.calculateRSI(prices2, 14);
        const trend1 = P1.detectTrend(prices1);
        const trend2 = P1.detectTrend(prices2);
        const vol1 = P1.calculateVolatility(prices1);
        const vol2 = P1.calculateVolatility(prices2);
        const macd1 = P1.calculateMACD(prices1);
        const macd2 = P1.calculateMACD(prices2);
        const bb1 = P1.calculateBollingerBands(prices1);

        const momentum1 = checkMomentum(prices1, P1.CONFIG.momentumBars);

        if (cointegrationPValue > P1.CONFIG.cointegrationPValue) {
            P1.advancedMetrics[pair.id] = { cointegration: cointegrationPValue, ouTheta, kalmanBeta, orderFlow: netOrderFlow, vwapDev, halfLife, hurst, expectedTPTime: 0 };
            return { signal: 'none', reason: 'Failed cointegration', confidence: 0, zScore, correlation, hurst, rsi1, rsi2, trend1, trend2, vol1, vol2, timestamp: new Date() };
        }

        if (correlation < P1.CONFIG.minCorrelation) {
            return { signal: 'none', reason: 'Low correlation', confidence: 0, zScore, correlation, hurst, rsi1, rsi2, trend1, trend2, vol1, vol2, timestamp: new Date() };
        }

        if (hurst > P1.CONFIG.hurstTrending) {
            return { signal: 'none', reason: 'Trending market', confidence: 0, zScore, correlation, hurst, rsi1, rsi2, trend1, trend2, vol1, vol2, timestamp: new Date() };
        }

        if (vol1 > P1.CONFIG.volatilityThreshold || vol2 > P1.CONFIG.volatilityThreshold) {
            return { signal: 'none', reason: 'High volatility', confidence: 0, zScore, correlation, hurst, rsi1, rsi2, trend1, trend2, vol1, vol2, timestamp: new Date() };
        }

        if (Math.abs(zScore) < P1.CONFIG.minZScore) {
            return { signal: Math.abs(zScore) > 2.5 ? 'wait' : 'none', reason: Math.abs(zScore) > 2.5 ? 'Building position' : 'Z-Score too low', confidence: 0, zScore, correlation, hurst, rsi1, rsi2, trend1, trend2, vol1, vol2, timestamp: new Date() };
        }

        if (P1.currentTimeframe === '1m') {
            if (zScore > 0 && momentum1 > 0) {
                return { signal: 'none', reason: 'Conflicting momentum', confidence: 0, zScore, correlation, hurst, rsi1, rsi2, trend1, trend2, vol1, vol2, timestamp: new Date() };
            }
            if (zScore < 0 && momentum1 < 0) {
                return { signal: 'none', reason: 'Conflicting momentum', confidence: 0, zScore, correlation, hurst, rsi1, rsi2, trend1, trend2, vol1, vol2, timestamp: new Date() };
            }
        }

        let confidence = 0;
        if (Math.abs(zScore) >= P1.CONFIG.strongZScore) confidence += 35;
        else if (Math.abs(zScore) >= P1.CONFIG.minZScore) confidence += 25;
        if (correlation >= 0.90) confidence += 25;
        else if (correlation >= 0.85) confidence += 20;
        if (cointegrationPValue < 0.05) confidence += 20;
        else if (cointegrationPValue < 0.10) confidence += 15;
        if (hurst < 0.35) confidence += 15;
        else if (hurst < 0.40) confidence += 12;
        if (zScore > 0) {
            if (rsi1 > 70 && rsi2 < 30) confidence += 10;
            else if (rsi1 > 60) confidence += 5;
        } else {
            if (rsi1 < 30 && rsi2 > 70) confidence += 10;
            else if (rsi1 < 40) confidence += 5;
        }
        if ((zScore > 0 && netOrderFlow < -0.1) || (zScore < 0 && netOrderFlow > 0.1)) confidence += 10;
        if ((zScore > 0 && vwapDev > 0) || (zScore < 0 && vwapDev < 0)) confidence += 5;
        if ((zScore > 0 && macd1.histogram < 0) || (zScore < 0 && macd1.histogram > 0)) confidence += 5;
        if ((zScore > 0 && price1 > bb1.upper) || (zScore < 0 && price1 < bb1.lower)) confidence += 5;

        let signal = 'none';
        let reason = 'Conditions not met';
        let binarySignal = null;

        if (confidence >= P1.CONFIG.minConfidence) {
            if (zScore > 0) {
                binarySignal = `${pair.asset1} DOWN`;
                signal = 'binary-down-1';
            } else {
                binarySignal = `${pair.asset1} UP`;
                signal = 'binary-up-1';
            }

            reason = `${binarySignal} | Conf: ${confidence}%`;
            P1.lastSignalTime[pair.id] = now;

            if (P1.signalTimers[pair.id]) clearTimeout(P1.signalTimers[pair.id]);
            P1.signalTimers[pair.id] = setTimeout(() => {
                if (P1.pairData[pair.id] && P1.pairData[pair.id].signal.startsWith('binary-')) {
                    P1.pairData[pair.id].signal = 'none';
                    P1.pairData[pair.id].reason = 'Signal expired';
                    delete P1.activeSignals[pair.id];
                    updateUI();
                }
            }, P1.CONFIG.signalDisplayTime);
        } else {
            signal = 'wait';
            reason = `Low confidence: ${confidence}%`;
        }

        const expiryRange = P1.EXPIRY_MAP[P1.currentTimeframe] || { min: 900, max: 1800 };
        const expectedTPTime = Math.floor(expiryRange.min + Math.random() * (expiryRange.max - expiryRange.min));

        P1.advancedMetrics[pair.id] = {
            cointegration: cointegrationPValue, ouTheta, kalmanBeta, orderFlow: netOrderFlow,
            vwapDev, halfLife, hurst, expectedTPTime, macd1: macd1.histogram, macd2: macd2.histogram,
            binarySignal
        };

        return { signal, binarySignal, reason, confidence, zScore, correlation, rsi1, rsi2, trend1, trend2, vol1, vol2, timestamp: new Date(), expectedTPTime };
    }

    function createBinaryTrade(pair, signalData) {
        const entryPrice = P1.priceHistory[pair.id].prices1[P1.priceHistory[pair.id].prices1.length - 1];
        const direction = signalData.signal.includes('up') ? 'UP' : 'DOWN';
        const asset = signalData.binarySignal.split(' ')[0];

        const trade = {
            id: Date.now(),
            pairId: pair.id,
            pairName: pair.name,
            asset: asset,
            direction: direction,
            signal: signalData.binarySignal,
            entryPrice: entryPrice,
            entryTime: Date.now(),
            expiryTime: Date.now() + (signalData.expectedTPTime * 1000),
            confidence: signalData.confidence,
            zScore: signalData.zScore,
            status: 'open',
            pnl: 0,
            exitPrice: null
        };

        P1.activeTrades[trade.id] = trade;

        setTimeout(() => {
            closeBinaryTrade(trade.id);
        }, signalData.expectedTPTime * 1000);

        return trade;
    }

    function closeBinaryTrade(tradeId) {
        const trade = P1.activeTrades[tradeId];
        if (!trade || trade.status !== 'open') return;

        const currentPrice = P1.priceHistory[trade.pairId].prices1[P1.priceHistory[trade.pairId].prices1.length - 1];

        trade.exitPrice = currentPrice;
        trade.status = 'closed';

        const priceChange = currentPrice - trade.entryPrice;

        if (trade.direction === 'UP') {
            trade.pnl = priceChange > 0 ? 85 : -100;
            trade.won = priceChange > 0;
        } else {
            trade.pnl = priceChange < 0 ? 85 : -100;
            trade.won = priceChange < 0;
        }

        addToHistory(trade);
        delete P1.activeTrades[tradeId];
        saveHistory();
    }

    function monitorActiveTrades() {
        Object.values(P1.activeTrades).forEach(trade => {
            if (trade.status === 'open') {
                const timeRemaining = trade.expiryTime - Date.now();
                if (timeRemaining <= 0) {
                    closeBinaryTrade(trade.id);
                }
            }

    function calculateHistoryStats() {
        if (P1.signalHistory.length === 0) return { winRate: 0, totalPnL: 0, totalSignals: 0, wins: 0, losses: 0 };

        let wins = 0, losses = 0, totalPnL = 0;

        P1.signalHistory.forEach(entry => {
            if (entry.status === 'closed') {
                totalPnL += entry.pnl;
                if (entry.won) wins++;
                else losses++;
            }
        });

        const totalTrades = wins + losses;

        return {
            winRate: totalTrades > 0 ? (wins / totalTrades) * 100 : 0,
            totalPnL: totalPnL,
            totalSignals: P1.signalHistory.length,
            wins: wins,
            losses: losses
        };

    function getBestPerformingPair() {

        P1.signalHistory.forEach(entry => {
            if (!pairStats[entry.pairName]) {
                pairStats[entry.pairName] = { wins: 0, total: 0 };
            }
            pairStats[entry.pairName].total++;
            if (entry.won) pairStats[entry.pairName].wins++;

        let bestPair = null;
        let bestWinRate = 0;

        for (const [pair, data] of Object.entries(pairStats)) {
